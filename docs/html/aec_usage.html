AEC 如何呼叫使用 — AEC/c_impl

概要

單聲道 AEC(Aec)一個實例處理一支 mic 與一路 ref。整合者只需要 aec.h;要在 AEC 之外算殘留回音抑制(RES)增益時再加 aec_res.h。本頁給呼叫順序、最小可編譯範例、初始化／對齊／錯誤處理／reset／destroy 規則;時序圖見 aec_sequence,參數見 外部參數 與 內部參數。下列範例 A–D(以及 aec_res 的獨立 RES 範例)已在暫存複本以 -Wall -Wextra -Werror 對 aec.h／aec_res.h 編譯,並在靜音輸入上跑完(含 aec_process 與 aec_process_context 兩條路),識別字全部存在;app_read_hop 等是呼叫端自己的音訊 I/O。

1. Include 與連結

#include "aec.h"        /* AEC/c_impl/include/aec.h:標準整合只需要這一個 */
#include "aec_res.h"    /* 只有自己建 RES 時才需要(4ch 應用在單元之後;見 aec_res.html) */

cc -std=gnu99 -O2 -ffp-contract=off app.c \
   -I<AEC>/c_impl/include -I<audiolib>/common/include \
   "$(make -s -C <AEC>/c_impl print-lib-path)" "$(make -s -C <audiolib>/common print-lib-path)" -lm -o app

libaec.a 不含共用 DSP,libaudio_common.a 一定要一起連結;-ffp-contract=off 是必要旗標(Makefile 會 append 在最後,見 AEC API §11)。aec.h 會 include audiolib 的 fft_wrapper.h／hpf.h,所以兩個 include 路徑都要給。輸出路徑含設定雜湊,用 print-lib-path 取,不要自己拼。

2. 呼叫順序

#步驟函式失敗時
1取得完整設定(不要手填)aec_config_from_preset(&cfg, preset, sample_rate)(或 aec_config_defaults)無回傳;不認得的 preset 值靜默退回 balanced
2只覆寫外部欄位(fft_size、delay_mode…)直接寫 cfg.*下一步才會被驗證
3(選用)確認格點aec_resolve_signal_grid(sr, fft, &grid)、aec_default_fft_size(sr)回 0(不動 *out)
4查記憶池大小(pool 路徑)aec_get_mem_size(&cfg);拆解用 aec_get_mem_breakdown回 0(不是負值)
5建立實例pool:aec_init(mem, bytes, &cfg);heap:aec_create(&aec, &cfg)aec_init 回 NULL(cfg 無效、未 16-byte 對齊、pool 過小);aec_create 回 -1
6取每次呼叫的樣本數aec_hop_size(aec)NULL 會 crash
7每個 hop 處理一次aec_process(aec, mic, ref, out) 或 aec_process_context(aec, mic, ref)回傳 void,沒有 per-call 錯誤碼;長度由呼叫端保證
8(選用)讀 seam／狀態aec_get_res_context、aec_get_linear_context、aec_debug_status、aec_linear_is_cancellingaec_get_*／aec_debug_status:NULL 實例或 NULL 輸出時靜默 no-op(不寫你的結構);aec_linear_is_cancelling 對 NULL 回 0
9(選用)hop 之間調整aec_set_preset(aec, preset, ramp_ms);aec_apply_external_realign(僅 EXTERNAL_ALIGNED)回 -1 且什麼都沒寫
10串流邊界aec_reset(aec)NULL 會 crash
11收尾aec_destroy(aec),pool 由呼叫端自己回收NULL 安全

3. 範例

A. 呼叫端記憶池(板端路徑,零 heap)

#include <stdlib.h>
#include "aec.h"

/* The application's own audio I/O: fills one hop of mic/ref, returns 0 at end of stream. */
extern int  app_read_hop(float *mic, float *ref, int hop);
extern void app_write_hop(const float *out, int hop);

int run_aec_pool(int sample_rate)
{
    AecConfig cfg;
    aec_config_from_preset(&cfg, AEC_PRESET_BALANCED, sample_rate);
    /* Override only external fields here, e.g. cfg.fft_size = 512 (16 kHz). */

    size_t bytes = aec_get_mem_size(&cfg);            /* 0 = cfg rejected */
    if (bytes == 0) return -1;
    void *pool = NULL;
    if (posix_memalign(&pool, 16, bytes) != 0) return -1;   /* base must be 16-byte aligned */

    Aec *aec = aec_init(pool, bytes, &cfg);           /* NULL = bad cfg / unaligned / too small */
    if (!aec) { free(pool); return -1; }

    int hop = aec_hop_size(aec);                      /* exactly this many samples per call */
    float *mic = malloc((size_t)hop * sizeof *mic);
    float *ref = malloc((size_t)hop * sizeof *ref);
    float *out = malloc((size_t)hop * sizeof *out);
    int rc = (mic && ref && out) ? 0 : -1;

    while (rc == 0 && app_read_hop(mic, ref, hop)) {
        aec_process(aec, mic, ref, out);              /* void: no per-call error code */
        app_write_hop(out, hop);
    }

    free(mic); free(ref); free(out);
    aec_destroy(aec);                                 /* pool instance: no-op, the pool stays yours */
    free(pool);
    return rc;
}

B. Heap(桌面／測試)

#include "aec.h"

int run_aec_heap(void)
{
    AecConfig cfg;
    Aec aec;                                          /* the struct is owned by the caller */
    aec_config_from_preset(&cfg, AEC_PRESET_BALANCED, 48000);
    if (aec_create(&aec, &cfg) != 0) return -1;       /* -1: cfg rejected or allocation failed */

    /* ... aec_process(&aec, mic, ref, out) once per hop ... */

    aec_destroy(&aec);                                /* frees the single internal arena: exactly once */
    return 0;
}

C. hop 之間的控制呼叫

#include "aec.h"

/* Calls made between two hops, serialised with aec_process*(); not thread-safe. */
int control_between_hops(Aec *aec, int stream_boundary, int user_wants_stronger)
{
    if (user_wants_stronger) {
        /* Walk the far-active floor to the new preset over 100 ms; -1 = rejected, nothing written. */
        if (aec_set_preset(aec, AEC_PRESET_AGGRESSIVE, 100.0f) != 0) return -1;
    }
    if (stream_boundary) {
        aec_reset(aec);          /* independent stream next: clears runtime state, keeps cfg and memory */
    }
    return 0;
}

/* Only for delay_mode == AEC_DELAY_EXTERNAL_ALIGNED: the caller changed its far alignment by delta_samples. */
int external_alignment_changed(Aec *aec, int delta_samples)
{
    int r = aec_apply_external_realign(aec, delta_samples);
    return r;                    /* 1 = learned IR shifted, 0 = filter reset, -1 = NULL or wrong delay_mode */
}

D. 只拿線性結果與頻域 seam(後級在下游)

#include <stdlib.h>
#include "aec.h"

extern int  app_read_hop(float *mic, float *ref, int hop);
extern void app_consume_frame(const AecResContext *ctx, const AecLinearContext *lin);

/* Linear AEC only; the residual-echo stage is done downstream from the frequency-domain seam. */
int run_aec_seam(int sample_rate)
{
    AecConfig cfg;
    Aec aec;
    aec_config_from_preset(&cfg, AEC_PRESET_BALANCED, sample_rate);
    cfg.enable_res         = 0;      /* no suppression applied inside the library */
    cfg.return_res_context = 1;      /* but the post block still runs and fills the context */
    if (aec_create(&aec, &cfg) != 0) return -1;

    int hop = aec_hop_size(&aec);
    float *mic = malloc((size_t)hop * sizeof *mic);
    float *ref = malloc((size_t)hop * sizeof *ref);
    int rc = (mic && ref) ? 0 : -1;

    while (rc == 0 && app_read_hop(mic, ref, hop)) {
        AecResContext ctx;
        AecLinearContext lin;
        aec_process_context(&aec, mic, ref);          /* no out buffer */
        aec_get_res_context(&aec, &ctx);              /* error_spec, r2, comfort_noise, res_gain, ... */
        aec_get_linear_context(&aec, &lin);           /* aligned_far_hop, delay_state, generation */
        app_consume_frame(&ctx, &lin);                /* read now: pointers die at the next process call */
    }

    free(mic); free(ref);
    aec_destroy(&aec);
    return rc;
}
seam 路徑要 同時 enable_res = 0 且 return_res_context = 1:只設前者,post 區塊整段不跑,error_spec／res_gain／r2／comfort_noise／formed_hop 全是 NULL(aec.c aec_get_res_context)。enable_cng 只管 enable_res = 1 時時域輸出內的舒適噪音注入;ctx.comfort_noise(PSD)只要 post 區塊有跑就會算,與 enable_cng 無關。AecResContext／AecLinearContext 的指標在下一次任何 process 或 reset 呼叫後失效,要跨 hop 就自己複製。

4. 初始化、對齊、錯誤處理、reset、destroy

項目規則(出處)
輸入長度mic／ref／out 各恰好 aec_hop_size() 個 float32,標稱 [-1, 1];處理函式沒有長度參數、不檢查(給少＝讀未初始化記憶體)。mic、ref、out 可互相 alias(aec_process_core 開頭就複製進內部 buffer)。
池對齊與大小基底 16-byte 對齊,aec_init 在寫任何 byte 前檢查(aec.c aec_init);mem_size 少 1 byte 即 NULL。Aec 結構本身也在池內。大小與格點、backend、struct 大小有關,永遠以 aec_get_mem_size() 為準。
查大小與 init 用同一份 cfg兩者各自驗證與計算,傳不同 cfg 不會被偵測(只擋 mem_size < need)。
init-time immutablesample_rate、fft_size、filter_length、n_partitions、delay_mode、delay_num_filters、fixed_delay_samples 決定 pool 佈局;改了要重新查大小並重建。aec_create／aec_init 會複製一份設定,事後改手上那份無效,也不要改 aec.cfg。
mic／ref 前提ref 是實際送往喇叭的訊號,與 mic 同取樣率同時鐘;延遲只能是正的(mic 落後 ref)。函式庫不補償 clock drift。
延遲三態MATCHED(預設,線上估計)／FIXED(給量測延遲 fixed_delay_samples)／EXTERNAL_ALIGNED(呼叫端保證已對齊);非法組合在建立時被拒絕,不會被修正。表見 AEC API §4。
執行緒一個實例不是 thread-safe;aec_process*、aec_set_preset、aec_reset、aec_apply_external_realign、aec_destroy 都要由呼叫端序列化。不同實例彼此獨立。
aec_reset清延遲估計、參考環、濾波器、偵測器、輸出 OLA 等執行期狀態,保留設定、記憶體與 aec_set_preset 設的目標;會讓 generation +1。每個獨立串流前呼叫;只是外部對齊改變時改用 aec_apply_external_realign(不重啟 OLA,輸出連續)。
aec_destroyheap 實例釋放單一 arena,剛好一次;pool 實例不釋放任何記憶體(只清內部參考)、可重複呼叫(NULL 也安全),池由呼叫端回收。
零 heappool 路徑從 init 到 destroy 不呼叫 malloc(make test-zero-heap:50 個 init/process/reset/destroy 週期 allocator 呼叫為 0)。heap 路徑只在 aec_create 做一次 16-byte 對齊配置。

各函式對 NULL 的行為(會 crash／安全／no-op)的完整表在 AEC/docs/c_user_manual_zh_TW_01_lifecycle.md §5。

5. 接到 NR 與合成(單聲道整鏈)

AEC 庫本身不包 NR 或合成。整合者各自初始化 Aec、AnrDenoiser(NR 庫)、AcIstft(audiolib),並自己寫一小段每 hop 的串接;Audio_ALG 的 pipelines/mono_aec_nr_res/main.c 就是這樣(串接函式 mono_form())。AEC 端以 enable_res = 0、return_res_context = 1 建立,單聲道的 RES 增益是 AEC 自己算的 ctx.res_gain(經 se_frame_from_aec() 成為 frame.g_res_ctx),不需另建 AecRes。

aec_process_context(aec, mic, ref);          /* 線性 + post 區塊,不寫 out */
se_frame_from_aec(aec, &frame);              /* E, R2, CN, frame.g_res_ctx(借用 AEC 的 buffer) */
g_nr = se_anr_stage_gain(nr, &frame, scratch, use_r2);   /* NR 增益;R2 併入噪音底(usable_linear 為真時) */
sk_min_f32(total, g_nr, frame.g_res_ctx, n_freqs);             /* g = min(g_nr, g_res) */
sk_capply_gain_f32(spec, frame.e, total, n_freqs);             /* P = E * g */
aec_cng_fill(&rng, spec, frame.cn, frame.g_res_ctx, g_nr, 0.31622777f, n_freqs);  /* 開 CNG 且有 RES 增益時 */
ac_istft_push(istft, spec, out);                               /* 反 FFT + 合成 + OLA */

上列後五行的名稱屬 Audio_ALG／audiolib,不是 AEC 庫的 API;流程時序見 aec_sequence §3,各 stage 的介面見 串接範例。
